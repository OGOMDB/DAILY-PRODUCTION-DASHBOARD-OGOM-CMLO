        const SHEET_ID = '1Ggjr0VeEEzBIJphWyyCYRQS5noukQz_ydI4Kauom7n4';

        function parseCSV(text) {
            const rows = [];
            let row = [];
            let cell = '';
            let inQuotes = false;

            for (let i = 0; i < text.length; i++) {
                const c = text[i];
                const next = text[i + 1];

                if (c === '"' && inQuotes && next === '"') {
                    cell += '"';
                    i++;
                } else if (c === '"') {
                    inQuotes = !inQuotes;
                } else if (c === ',' && !inQuotes) {
                    row.push(cell);
                    cell = '';
                } else if (c === '\n' && !inQuotes) {
                    row.push(cell);
                    if (row.length > 0) rows.push(row);
                    row = [];
                    cell = '';
                } else {
                    cell += c;
                }
            }

            if (cell || row.length > 0) {
                row.push(cell);
                rows.push(row);
            }

            return rows;
        }

        async function fetchSheet(sheetName) {
            const timestamp = new Date().getTime();
            const url = `https://docs.google.com/spreadsheets/d/${SHEET_ID}/gviz/tq?tqx=out:csv&sheet=${encodeURIComponent(sheetName)}&t=${timestamp}`;

            const response = await fetch(url, {
                method: 'GET',
                cache: 'no-store',
                headers: {
                    'Cache-Control': 'no-cache',
                    'Pragma': 'no-cache'
                }
            });

            if (!response.ok) throw new Error(`Failed to fetch ${sheetName}`);
            return parseCSV(await response.text());
        }

        async function loadAllData() {
            const statusEl = document.getElementById('status');
            try {
                statusEl.className = 'status loading';
                statusEl.textContent = '⏳ Loading data from Google Sheets...';

        }

        function extractKPIs(data) {
            let totalProduction = 0, lumpPercentage = 0, rakesDispatched = 0, dailyCODPlan = 0;
            let productionDate = '', reportingDate = '';

            console.log('🔍 Extracting KPIs and dates...');

            // Find dates - search first 40 rows
            for (let i = 0; i < Math.min(40, data.length); i++) {
                const row = data[i];
                const rowText = row.join(' ');

                for (let j = 0; j < row.length; j++) {
                    const cell = (row[j] || '').toString();
                    const cellLower = cell.toLowerCase();

                    // Date pattern
                    const dateMatch = cell.match(/(\d{1,2}[\/\-]\d{1,2}[\/\-]\d{4}(\s+\d{1,2}:\d{2}(:\d{2})?)?)/);

                    if (dateMatch) {
                        // Production Date
                        if (!productionDate && (cellLower.includes('production') || rowText.toLowerCase().includes('production date'))) {
                            productionDate = dateMatch[1].split(' ')[0];
                            console.log(`✅ Production Date: "${productionDate}" at [${i},${j}]`);
                        }

                        // Report Generated
                        if (!reportingDate && (cellLower.includes('report') || cellLower.includes('generat') || rowText.toLowerCase().includes('report generat'))) {
                            reportingDate = dateMatch[1];
                            console.log(`✅ Report Generated: "${reportingDate}" at [${i},${j}]`);
                        }
                    }
                }

                // Find KPI row (large number > 10000)
                const col0Num = parseFloat((row[0] || '').toString().replace(/,/g, ''));
                if (col0Num > 10000 && totalProduction === 0) {
                    totalProduction = col0Num;
                    dailyCODPlan = parseFloat((row[1] || '').toString().replace(/,/g, '')) || 0;
                    lumpPercentage = parseFloat((row[2] || '').toString().replace(/,/g, '')) || 0;
                    rakesDispatched = parseFloat((row[3] || '').toString().replace(/,/g, '')) || 0;
                    console.log(`✅ KPIs found at row ${i}`);
                }
            }

            // Update UI
            document.getElementById('prod-date').textContent = `Production Date: ${productionDate || 'NOT FOUND'}`;
            document.getElementById('report-date').textContent = `Report Generated: ${reportingDate || 'NOT FOUND'}`;

            const planAchievement = dailyCODPlan > 0 ? (totalProduction / dailyCODPlan) * 100 : 0;

            return { totalProduction, planAchievement, lumpPercentage, rakesDispatched, dailyCODPlan, productionDate, reportingDate };
        }

        function extractProduction(data) {
            const production = [];
            let inSection = false;

            for (let i = 0; i < data.length; i++) {
                const row = data[i];
                const col0 = (row[0] || '').toString().trim();

                if (col0.toLowerCase().includes('production performance') || col0.toLowerCase().includes('mine/area')) {
                    inSection = true;
                    continue;
                }

                if (inSection && (col0.toLowerCase().includes('dispatch') || col0.toLowerCase().includes('loading'))) {
                    break;
                }

                if (inSection && col0 && col0.match(/Bolani|Barsua|Taldih|Kalta|OGISHA GROUP/i)) {
                    const type = (row[1] || '').toString().trim();
                    const total = parseFloat((row[2] || '').toString().replace(/,/g, '')) || 0;
                    const lump = parseFloat((row[3] || '').toString().replace(/,/g, '')) || 0;
                    const fines = parseFloat((row[4] || '').toString().replace(/,/g, '')) || 0;
                    const rom = parseFloat((row[5] || '').toString().replace(/,/g, '')) || 0;
                    const slime = parseFloat((row[6] || '').toString().replace(/,/g, '')) || 0;
                    const ob = parseFloat((row[7] || '').toString().replace(/,/g, '')) || 0;
                    const lumpPercent = parseFloat((row[8] || '').toString().replace(/,/g, '')) || 0;
                    const plan = parseFloat((row[9] || '').toString().replace(/,/g, '')) || 0;
                    const achievement = parseFloat((row[10] || '').toString().replace(/,/g, '')) || 0;

                    production.push({
                        mine: col0,
                        type,
                        total,
                        lump,
                        fines,
                        rom,
                        slime,
                        ob,
                        lumpPercent,
                        plan,
                        achievement,
                        isTotal: col0.toUpperCase().includes('TOTAL') || col0.toUpperCase().includes('GROUP')
                    });
                }
            }

            console.log(`✅ Extracted ${production.length} production rows`);
            return production;
        }

        function extractDispatch(data) {
            const dispatch = [];
            let startRow = -1;
            let endRow = -1;

            // Find section boundaries
            for (let i = 0; i < data.length; i++) {
                const rowText = data[i].join('|').toUpperCase();

                if (rowText.includes('DISPATCH PERFORMANCE') && startRow === -1) {
                    startRow = i;
                    console.log(`📊 DISPATCH starts at row ${i}`);
                }

                if (startRow > -1 && i > startRow + 1 && (rowText.includes('LOADING FORECAST') || rowText.includes('MAJOR ISSUES') || rowText.includes("TODAY'S"))) {
                    endRow = i;
                    console.log(`📊 DISPATCH ends at row ${i}`);
                    break;
                }
            }

            if (startRow === -1) {
                console.error('❌ DISPATCH section not found');
                return dispatch;
            }

            // Extract data ONLY from this range
            for (let i = startRow; i < (endRow > -1 ? endRow : data.length); i++) {
                const row = data[i];
                const col0 = (row[0] || '').toString().trim();

                if (['BOL', 'BAR', 'TAL', 'KAL', 'TOTAL'].includes(col0)) {
                    console.log(`📊 DISPATCH Row ${i}: ${col0}`);

                    const rakes = parseFloat((row[1] || '').toString().replace(/,/g, '')) || 0;
                    const dailyCOD = parseFloat((row[2] || '').toString().replace(/,/g, '')) || 0;
                    const achievement = parseFloat((row[3] || '').toString().replace(/,/g, '')) || 0;
                    const rakeDetails = (row[4] || '').toString().trim();
                    const remarks = (row[5] || '').toString().trim();

                    console.log(`  Rake Details: "${rakeDetails.substring(0, 40)}..."`);
                    console.log(`  Remarks: "${remarks.substring(0, 40)}..."`);

                    dispatch.push({
                        location: col0,
                        rakes,
                        dailyCOD,
                        achievement,
                        rakeDetails,
                        remarks
                    });
                }
            }

            console.log(`✅ Extracted ${dispatch.length} dispatch rows`);
            return dispatch;
        }

        function extractForecast(data) {

                    break;
                }
            }

            if (startRow === -1) {
                console.error('❌ FORECAST section not found');

                const row = data[i];

        function extractIssues(data) {
            const issues = [];
            let inSection = false;

            for (let i = 0; i < data.length; i++) {
                const row = data[i];
                const col0 = (row[0] || '').toString().trim();

                if (col0.toLowerCase().includes('major issues')) {
                    inSection = true;
                    continue;
                }

                if (inSection && col0.toLowerCase().includes('loading forecast')) {
                    break;
                }

                if (inSection && col0.match(/Bolani|Barsua|Taldih|Kalta/i)) {
                    issues.push({
                        mine: col0,
                        issue: (row[1] || '').toString().trim()
                    });
                }
            }

            console.log(`✅ Extracted ${issues.length} issues`);
            return issues;
        }
